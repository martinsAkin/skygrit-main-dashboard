 import { type ModulesProps } from "../../interface";

     setVisible(!visible);

               >
                 <option value="">Select Management Type</option>

                           placeholder="Enter client name"

                         Email Address
                       </label>
                       <input
                         type="text"
                         name="email"
                         required
                         placeholder="Enter address"
                         className="inputStyling"
                       />

                       <label htmlFor="businessUnit" className="labelStyling">
                         Business Unit
                       </label>
                       <span className="relative flex flex-col">

                           id="businessUnit"
                           name="businessUnit"
                           className="inputStyling"
                           value={businessUnit}
                           onChange={(e) => setBusinessUnit(e.target.value)}

                           {businessUnitOption.map((type, index) => (
                             <option key={index} value={type}>
                               {type}
                             </option>
                           ))}
                         </select>
                       </span>
                     </div>

                     <div className="flex flex-col gap-1">
                       <label htmlFor="clientType" className="labelStyling">
                         Client Type
                       </label>
                       <span className="relative flex flex-col">
                         <select
                           id="clientType"
                           name="clientType"
                           value={clientType}
                           onChange={(e) => setClientType(e.target.value)}
                           className="inputStyling"
                         >
                           <option value="">--Select Client Type--</option>
                           {clientTypeOptions.map((type, index) => (
                             <option key={index} value={type.value}>
                               {type.option}
                             </option>
                           ))}
                         </select>
                       </span>
                     </div>
                 </div>
                 )
                 :
                 mgtType == "general" ? (
                   <div>
                       <div className="flex flex-col gap-1">
                       <label htmlFor="businessUnit" className="labelStyling">
                         Business Unit
                       </label>
                       <span className="relative flex flex-col">
                         <select
                           id="businessUnit"
                           name="businessUnit"
                           className="inputStyling"
                           value={businessUnit}
                           onChange={(e) => setBusinessUnit(e.target.value)}
                         >
                           <option value="">--Select Business Unit--</option>
                           {businessUnitOption.map((type, index) => (
                             <option key={index} value={type}>
                               {type}
                             </option>
                           ))}
                         </select>
                       </span>
                     </div>

                     <div className="flex flex-col gap-1">
                       <label htmlFor="clientType" className="labelStyling">
                         Client Type
                       </label>
                       <span className="relative flex flex-col">
                         <select
                           id="clientType"
                           name="clientType"
                           value={clientType}
                           onChange={(e) => setClientType(e.target.value)}
                           className="inputStyling"
                         >
                           <option value="">--Select Client Type--</option>
                           {clientTypeOptions.map((type, index) => (
                             <option key={index} value={type.value}>
                               {type.option}
                             </option>
                           ))}

                       </span>

                   <div>
                     <select
                         value={clientCategory}
                         onChange={() => setClientCategory("corporate")}
                         className="inputStyling"
                       >
                         <option value="corporate">Corporate</option>

                       </label>
                       <input
                         type="text"
                         name="account"
                         required
                         placeholder="Enter Account"
                         className="inputStyling"
                       />
                     </div>
                   </div>
                  ) : ""}

                 {
                  mgtType == "specific" && clientCategory == "corporate" ? (
                     <div className="flex flex-col gap-1">
                       <label htmlFor="account" className="labelStyling">
                         Internal Account
                       </label>
                       <input
                         type="text"
                         name="account"
                         required
                         placeholder="Enter Account"
                         className="inputStyling"
                       />
                     </div>
                   )
                   : 
                   mgtType == "specific" && clientCategory == "nonCorporate" ? (
                     <div>
                         <select
                         value={fares}
                         onChange={(e) => setFares(e.target.value)}
                         className="inputStyling"
                       >
                         <option value="regular">Regular</option>

                   ) : ""

                   mgtType == "general" && clientType == "directSales" ? (
                     <div>
                       <select
                         value={clientCategory}
                         onChange={(e) => setClientCategory(e.target.value)}
                         className="inputStyling"
                       >
                         <option value="corporate">Select Client Category</option>
                         <option value="corporate">Corporate</option>

                     mgtType == "general" && clientType == "indirectSales" ? (
                     <div>
                       <select
                         value={clientCategory}
                         onChange={() => setClientCategory("corporate")}
                         className="inputStyling"
                       >

                       </select>

                         </label>
                         <input
                           type="text"
                           name="email"
                           required
                           placeholder="Enter address"
                           className="inputStyling"
                         />
                       </div>

                           Internal Account

                         <input
                           type="text"
                           name="account"
                           required
                           placeholder="Enter Account"
                           className="inputStyling"
                         />
                       </div>

                     </div>
                     )
                     : ""

                 {
                   mgtType == "general" && clientCategory == "corporate" ? (
                     <div>

                           Email Address
                         </label>
                         <input
                           type="text"
                           name="email"
                           required
                           placeholder="Enter address"
                           className="inputStyling"
                         />
                       </div>

                       <div className="flex flex-col gap-1">
                         <label htmlFor="account" className="labelStyling">
                           Internal Account
                         </label>
                         <input
                           type="text"
                           name="account"
                           required
                           placeholder="Enter Account"
                           className="inputStyling"
                         />
                       </div>
                     </div>
                   ) 
                   : 
                     mgtType == "general" && clientCategory == "nonCorporate" ? (
                     <div>
                       <select
                         value={fares}
                         onChange={(e) => setFares(e.target.value)}
                         className="inputStyling"
                       >
                         <option value="regular">Regular</option>
                         <option value="discounted">Discounted/Promotional</option>
                       </select>
                     </div>
                     )
                     : ""
                 }

             </div>
           </form>

           <ModulesBtnSet onCancel={onCancel} text2="Create User"/>
         </div>
       </div>
     </div>
   );
 };

 export default AddNewClient;
